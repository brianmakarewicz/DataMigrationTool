Public Function GetFileBytes(ByVal path As Variant) As Byte()
    Dim lngFileNum As Long
    Dim bytRtnVal() As Byte
    lngFileNum = FreeFile
    If LenB(Dir(path)) Then ''// Does file exist?
        Open path For Binary Access Read As lngFileNum
        ReDim bytRtnVal(LOF(lngFileNum) - 1&) As Byte
        Get lngFileNum, , bytRtnVal
        Close lngFileNum
    Else
        Err.Raise 53
    End If
    GetFileBytes = bytRtnVal
    Erase bytRtnVal
End Function

Public Function WriteBToSToBFile(fileSaveName As Variant, readFrom As Variant)
    Dim byteA() As Byte
             
    byteA = GetFileBytes(readFrom)
     
    Dim ds As String
    Dim ts As String
     
    ds = byteA
    ts = Left(ds, 2)
    ds = Right(ds, Len(ds) - 1)
         
    ds = VBA.Replace(ds, ChrW(9), ChrW(44))
 
     
    ds = VBA.Replace(ds, ChrW(34) & ChrW(34) & ChrW(34), ChrW(34))
    
    'ds = Replace(ds, "START", "") Commented for Bug37383756
    
 
    Dim fsT
    Set fsT = CreateObject("ADODB.Stream")
    fsT.Type = 2
    fsT.Charset = "UTF-8"
    fsT.Open
     
    fsT.WriteText ds
 
    'the fix starts here (Removal of BOM)
 
    fsT.Position = 3 'skip BOM
    Dim BinaryStream As Object
    Set BinaryStream = CreateObject("ADODB.Stream")
    BinaryStream.Type = 1
    BinaryStream.Open
    'Strips BOM (first 3 bytes)
    fsT.CopyTo BinaryStream
    'fsT.SaveToFile fileSaveName, 2
    BinaryStream.SaveToFile fileSaveName, 2
    BinaryStream.flush
    BinaryStream.Close
 
    'the fix ends here (Removal of BOM)
     
    fsT.flush
    fsT.Close
End Function
